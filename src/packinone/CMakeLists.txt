# packinone - 個別プラグイン (src/plugins/*) を 1 つの DLL に詰め合わせる
#
# 各プラグインのソースは **そのまま** 取り込む (二重管理しない)。
# 取り込むソースには per-source で
#   TVP_STATIC_PLUGIN       … ncbind の自動登録リストを名前で分ける + 登録エントリを生やす
#   TVP_PLUGIN_NAME=<名前>  … その名前
#   V2Link / V2Unlink / DllEntryPoint … 名前衝突を避けるため改名
# を与える。 詰め合わせ側の V2Link は packinone.cpp が持つ。
#
# ⚠ ncbind を使うプラグインは V2Link と登録エントリが ncbind.cpp 側にあるので、
#   **プラグインごとに ncbind.cpp を別コンパイル**する (生成した薄いラッパ経由)。
#
# 詳細は README.md。

set(TPSTUB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../plugins/tp_stub CACHE PATH "Path to tp_stub directory")
if(NOT COMMAND krkrz_plugin)
    include(${TPSTUB_DIR}/krkrz.cmake)
endif()

set(PACKINONE_PLUGINS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../plugins)
set(PACKINONE_NCBIND_CPP  ${PACKINONE_PLUGINS_DIR}/ncbind/ncbind.cpp)
set(PACKINONE_SOURCES "")

# packinone_absorb(<名前> NCBIND <ソース...>)
function(packinone_absorb name)
    set(srcs ${ARGN})
    set(use_ncbind FALSE)
    if("NCBIND" IN_LIST srcs)
        set(use_ncbind TRUE)
        list(REMOVE_ITEM srcs NCBIND)
    endif()

    set(defs "TVP_STATIC_PLUGIN;TVP_PLUGIN_NAME=${name};V2Link=V2Link_${name};V2Unlink=V2Unlink_${name};DllEntryPoint=DllEntryPoint_${name}")
    set(incs "${PACKINONE_PLUGINS_DIR}/${name};${PACKINONE_PLUGINS_DIR}/ncbind;${PACKINONE_PLUGINS_DIR}")

    set(files "")
    foreach(src ${srcs})
        list(APPEND files ${PACKINONE_PLUGINS_DIR}/${name}/${src})
    endforeach()

    if(use_ncbind)
        # ncbind.cpp をプラグインごとにコンパイルするための薄いラッパを生成する
        set(wrapper ${CMAKE_CURRENT_BINARY_DIR}/ncbind_${name}.cpp)
        file(WRITE ${wrapper}
             "// generated by src/packinone/CMakeLists.txt\n"
             "#include \"${PACKINONE_NCBIND_CPP}\"\n")
        list(APPEND files ${wrapper})
    endif()

    set_source_files_properties(${files} PROPERTIES
        COMPILE_DEFINITIONS "${defs}"
        INCLUDE_DIRECTORIES "${incs}")

    set(PACKINONE_SOURCES ${PACKINONE_SOURCES} ${files} PARENT_SCOPE)
endfunction()

packinone_absorb(csvParser              main.cpp)
packinone_absorb(saveStruct    NCBIND   Main.cpp)
packinone_absorb(scriptsEx     NCBIND   Main.cpp)
packinone_absorb(shrinkCopy    NCBIND   main.cpp)
packinone_absorb(layerExBTOA   NCBIND   Main.cpp)
packinone_absorb(layerExRaster NCBIND   main.cpp)

krkrz_plugin(packinone
    INCLUDES
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${PACKINONE_PLUGINS_DIR}
        ${PACKINONE_PLUGINS_DIR}/ncbind
    SOURCES
        packinone.cpp
        ${PACKINONE_SOURCES}
)

# 歴史的な名前 (スクリプトは "PackinOne.dll" を link する)
set_target_properties(packinone PROPERTIES OUTPUT_NAME PackinOne)

# packinone - 個別プラグインのソースを 1 つの DLL に詰め合わせる
#
# 各プラグインのソースは **そのまま** 取り込む (二重管理しない)。
# 取り込むソースには per-source で
#   TVP_STATIC_PLUGIN       … ncbind / simplebinder の自動登録リストを名前で分ける
#                             + krkrz_plugin_<名前>() の登録エントリを生やす
#   TVP_PLUGIN_NAME=<名前>  … その名前
#   V2Link / V2Unlink / DllEntryPoint … 名前衝突を避けるため改名
# を与える。 詰め合わせ側の V2Link は packinone.cpp が持つ。
#
# ⚠ バインダ (ncbind.cpp / v2link.cpp) には V2Link と登録エントリが入っているので、
#   **プラグインごとに別コンパイル**する (生成した薄いラッパ経由)。
#
# 詳細は README.md。

set(TPSTUB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../plugins/tp_stub CACHE PATH "Path to tp_stub directory")
if(NOT COMMAND krkrz_plugin)
    include(${TPSTUB_DIR}/krkrz.cmake)
endif()

set(PACKINONE_PLUGINS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../plugins)
set(PACKINONE_NCBIND_CPP  ${PACKINONE_PLUGINS_DIR}/ncbind/ncbind.cpp)
set(PACKINONE_V2LINK_CPP  ${PACKINONE_PLUGINS_DIR}/simplebinder/v2link.cpp)
set(PACKINONE_SOURCES "")
set(PACKINONE_INCLUDES "")

# packinone_absorb(<名前>
#     [NCBIND | SIMPLEBIND]      使っているバインダ
#     [DIR      <path>]          ソースの置き場 (既定 ../plugins/<名前>)
#     [SOURCES  <src...>]        DIR からの相対パス
#     [INCLUDES <dir...>]        DIR からの相対パス、または絶対パス
#     [DEFINITIONS <def...>])
function(packinone_absorb name)
    cmake_parse_arguments(A "NCBIND;SIMPLEBIND" "DIR" "SOURCES;INCLUDES;DEFINITIONS" ${ARGN})

    if(NOT A_DIR)
        set(A_DIR ${PACKINONE_PLUGINS_DIR}/${name})
    endif()
    get_filename_component(A_DIR "${A_DIR}" ABSOLUTE)

    set(defs "TVP_STATIC_PLUGIN;TVP_PLUGIN_NAME=${name};V2Link=V2Link_${name};V2Unlink=V2Unlink_${name};DllEntryPoint=DllEntryPoint_${name}")
    list(APPEND defs ${A_DEFINITIONS})

    # ⚠ 取り込むプラグイン自身のヘッダを先に見せる (tp_stub / ncbind の古い写しを拾わないため)
    set(incs "${A_DIR}")
    foreach(inc ${A_INCLUDES})
        if(IS_ABSOLUTE "${inc}")
            list(APPEND incs "${inc}")
        else()
            list(APPEND incs "${A_DIR}/${inc}")
        endif()
    endforeach()
    list(APPEND incs
        "${PACKINONE_PLUGINS_DIR}/ncbind"
        "${PACKINONE_PLUGINS_DIR}/simplebinder"
        "${PACKINONE_PLUGINS_DIR}")

    set(files "")
    foreach(src ${A_SOURCES})
        list(APPEND files ${A_DIR}/${src})
    endforeach()

    # バインダはプラグインごとにコンパイルする必要があるので薄いラッパを生成する
    set(binder "")
    if(A_NCBIND)
        set(binder ${PACKINONE_NCBIND_CPP})
    elseif(A_SIMPLEBIND)
        set(binder ${PACKINONE_V2LINK_CPP})
    endif()
    if(binder)
        set(wrapper ${CMAKE_CURRENT_BINARY_DIR}/binder_${name}.cpp)
        file(WRITE ${wrapper}
             "// generated by src/packinone/CMakeLists.txt\n"
             "#include \"${binder}\"\n")
        list(APPEND files ${wrapper})
    endif()

    set_source_files_properties(${files} PROPERTIES
        COMPILE_DEFINITIONS "${defs}"
        INCLUDE_DIRECTORIES "${incs}")

    set(PACKINONE_SOURCES ${PACKINONE_SOURCES} ${files} PARENT_SCOPE)
endfunction()

packinone_absorb(csvParser     SOURCES main.cpp)
packinone_absorb(saveStruct    NCBIND SOURCES Main.cpp)
packinone_absorb(scriptsEx     NCBIND SOURCES Main.cpp)
packinone_absorb(shrinkCopy    NCBIND SOURCES main.cpp)
packinone_absorb(layerExBTOA   NCBIND SOURCES Main.cpp)
packinone_absorb(layerExRaster NCBIND SOURCES main.cpp)
packinone_absorb(layerExImage  NCBIND SOURCES LayerExImage.cpp Main.cpp LicensesGen.cpp)

# tjsDataPack だけは現行ツリーの外 (krkrtemplate/plugins_utf8) にある
set(PACKINONE_TJSDATAPACK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../krkrtemplate/plugins_utf8/tjsDataPack"
    CACHE PATH "Path to tjsDataPack sources")
if(EXISTS "${PACKINONE_TJSDATAPACK_DIR}/Main.cpp")
    packinone_absorb(tjsDataPack SIMPLEBIND
        DIR      ${PACKINONE_TJSDATAPACK_DIR}
        INCLUDES lz4 xxHash
        SOURCES
            Main.cpp
            VariantArchive.cpp
            LayerImage.cpp
            ImageChunk.cpp
            lz4/lz4.c
            lz4_license_bin.c
            xxHash/xxhash.c
            xxhash_license_bin.c)
    set(PACKINONE_HAS_TJSDATAPACK TRUE)
else()
    message(WARNING "packinone: tjsDataPack sources not found at ${PACKINONE_TJSDATAPACK_DIR} - skipped")
endif()

krkrz_plugin(packinone
    INCLUDES
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${PACKINONE_PLUGINS_DIR}
        ${PACKINONE_PLUGINS_DIR}/ncbind
    SOURCES
        packinone.cpp
        ${PACKINONE_SOURCES}
)

target_compile_features(packinone PRIVATE cxx_std_17)

if(PACKINONE_HAS_TJSDATAPACK)
    target_compile_definitions(packinone PRIVATE PACKINONE_HAS_TJSDATAPACK)
endif()

# 歴史的な名前 (スクリプトは "PackinOne.dll" を link する)
set_target_properties(packinone PROPERTIES OUTPUT_NAME PackinOne)
